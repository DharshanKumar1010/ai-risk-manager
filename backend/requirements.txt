# --- Data and ML layer -------------------------------------------------------
# Added in Phase 1 for the feature pipeline.
#
# econml was reserved for Phase 6 and was deliberately NOT added when Phase 6 shipped. Three
# reasons, in order of weight. Its published pins have capped scikit-learn below 1.6 and numpy
# below 2, against this file's sklearn>=1.9, numpy>=2.4 and pandas>=3.0. `filterwarnings =
# ["error"]` in pyproject turns any import-time deprecation into a collection failure, which is
# exactly how shap became a declared-but-unusable dependency below. And the cross-fitting here
# has to be chronological -- econml's folds are random, which on time-ordered transaction data
# is the leak the whole project is built to avoid.
#
# The DR/AIPW estimators are therefore hand-rolled over numpy in app/ml/ope.py, which is ~40
# lines of arithmetic, fully typed under mypy strict, and testable against exact ground truth.
#
# pandas 3.x is the current major: copy-on-write is enforced and string columns are
# PyArrow-backed by default, which matters directly here — train_transaction.csv is
# ~652MB with many low-cardinality string columns.
numpy>=2.4,<3.0
pandas>=3.0,<4.0
pyarrow>=25.0,<27.0
scikit-learn>=1.9,<2.0

# Phase 2, Tier-1. LightGBM is the supervised half of the Tier-1 comparison.
#
# `shap` is declared but deliberately never imported. Both tree layers take TreeSHAP from
# their own booster instead — LightGBM's `pred_contrib=True` (tier1_anomaly.explain) and
# XGBoost's `pred_contribs=True` (meta_learner.contributions). Phase 5 verified the latter
# is bit-identical to `shap.TreeExplainer.shap_values` (max abs diff 0.0) and that
# `import shap` raises PendingDeprecationWarning from matplotlib at import time, which our
# `filterwarnings = ["error"]` pytest config turns into a collection failure. Kept in the
# manifest only so pip-audit tracks it; remove it in Phase 10 if nothing has imported it by then.
lightgbm>=4.5,<5.0
shap>=0.46,<1.0

# Phase 3, Tier-2. The CPU build only: Tier-2's encoder is an LSTM, and cuDNN's LSTM
# kernel is non-deterministic, which would falsify the reproducibility claim
# models/registry.json makes about every model in it. The CPU wheel is what PyPI serves
# by default on Windows and macOS; on Linux the Dockerfile takes it from the explicit
# cpu index, so the image does not carry ~2GB of unused CUDA libraries.
torch>=2.5,<3.0

# Phase 3 requires the reconstruction-error distribution and the training loss curve as
# plots, and they gate the phase. Agg backend only — nothing here opens a window.
# Phase 4 reuses it for the detected-ring and clean-cluster visualisations.
matplotlib>=3.9,<4.0

# Phase 4, Tier-3. NetworkX over python-igraph: `networkx.community.louvain_communities`
# is built in, removing a `python-louvain` dependency, and igraph's speed advantage does not
# bind here. Phase 4 measured PaySim's money-flow graph to be a star forest of small
# components (max destination degree 71) and caps IEEE-CIS entity degree, so community
# detection runs per connected component and never over one large graph. The graph is held
# behind our own `EntityGraph` wrapper, so swapping the backend stays a backend change.
#
# NetworkX 3.6 does NOT ship `py.typed`, contrary to what the Phase 4 plan assumed — mypy
# reports missing stubs for it. Rather than widen the `ignore_missing_imports` allowlist in
# pyproject.toml, `types-networkx` is added below, for the same reason `pandas-stubs` is
# there: under strict mode an untyped graph library makes every expression touching a node,
# an edge or an attribute `Any`, and this is the layer where a leakage bug would hide.
networkx>=3.4,<4.0

# Phase 5, meta-learner. XGBoost over LightGBM for the fusion layer purely so the two tree
# libraries stay distinguishable in the registry and in SHAP output — the meta-learner
# consumes Tier-1's LightGBM score as a feature, and one library producing both would make a
# stack trace ambiguous. `n_jobs` is pinned to 4 in XGBOOST_PARAMS for the same determinism
# reason LightGBM's `num_threads` is: models/registry.json claims reproducibility for every
# model in it. Ships no py.typed, so pyproject.toml's ignore_missing_imports list covers it.
xgboost>=2.1,<4.0
